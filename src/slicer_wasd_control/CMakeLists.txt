cmake_minimum_required(VERSION 3.0.2)
project(slicer_wasd_control)
find_package(catkin REQUIRED COMPONENTS rospy geometry_msgs nav_msgs std_msgs tf2_ros)
catkin_package()
catkin_install_python(PROGRAMS scripts/c30d_driver_node.py scripts/velocity_mux.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(FILES scripts/driver_core.py scripts/control_core.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
set(C30D_HARDWARE "${CMAKE_CURRENT_SOURCE_DIR}/../../Hardware")
foreach(module chassis chassis_recovery chassis_commands protocol transport)
  install(FILES "${C30D_HARDWARE}/Host/${module}.py" DESTINATION Hardware/Host)
endforeach()
install(FILES "${C30D_HARDWARE}/Unlock/unlock.py" DESTINATION Hardware/Unlock)
install(FILES "${C30D_HARDWARE}/config.yaml" DESTINATION Hardware)
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
